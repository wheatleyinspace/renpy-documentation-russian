Стандартные фрагменты шейдеров

Фрагменты шейдеров используются в модельном рендеринге и текстовых шейдерах. Этот файл содержит указатель фрагментов по приоритетам и исходный код каждого фрагмента.

Приоритеты фрагментов вершинного шейдера

Использование приоритетов вершинных фрагментов в Ren'Py:

  • Приоритет 0 инициализирует gl_Position.

  • Приоритет 10 сохраняет gl_Position в виртуальных или физических координатах до корректировок.

  • Приоритеты 20-80 корректируют gl_Position в виртуальных или физических координатах.

  • Приоритет 90 сохраняет gl_Position после корректировок.

  • Приоритет 100 преобразует gl_Position в координаты viewport.

  • Приоритет 200 передаёт дополнительную информацию во фрагментный шейдер без изменения gl_Position.

Приоритеты фрагментов пиксельного (фрагментного) шейдера

Использование приоритетов фрагментных частей в Ren'Py:

  • Приоритет 200 определяет исходный цвет и записывает его в выход фрагмента (fragment_color в GLSL ES 3.00 или gl_FragColor в GLSL ES 1.00).

  • Приоритет 300 умножает этот цвет на вторую текстуру.

  • Приоритет 325 сохраняет значение альфы перед модификацией текстовыми шейдерами.

  • Приоритет 350 применяет текстовые шейдеры, изменяющие альфу.

  • Приоритет 375 может отменять часть эффекта этих текстовых шейдеров.

  • Приоритет 400 корректирует цвет на основе Transform и отображаемых объектов.

  • Приоритет 450 преобразует цвет в альфу.

  • Приоритет 500 корректирует альфа-канал на основе Transform и отображаемых объектов.

renpy.blur

Variables:

uniform sampler2D tex0;
uniform vec2 res0;
uniform float u_renpy_blur_log2;

in vec2 a_tex_coord;
out vec2 v_tex_coord;

out vec2 v_renpy_blur_lod_blend;
out vec4 v_renpy_blur_floor_size;
out vec4 v_renpy_blur_ceil_size;

Vertex shader (priority 200):

v_tex_coord = a_tex_coord;

float l_renpy_blur_lod = floor(u_renpy_blur_log2);

v_renpy_blur_lod_blend = vec2(l_renpy_blur_lod, u_renpy_blur_log2 - l_renpy_blur_lod);

vec2 l_renpy_blur_floor_size = max(vec2(1.0), floor(res0 / exp2(l_renpy_blur_lod)));
vec2 l_renpy_blur_ceil_size = max(vec2(1.0), floor(l_renpy_blur_floor_size * 0.5));

v_renpy_blur_floor_size = vec4(l_renpy_blur_floor_size, 1.0 / l_renpy_blur_floor_size);
v_renpy_blur_ceil_size = vec4(l_renpy_blur_ceil_size, 1.0 / l_renpy_blur_ceil_size);

Fragment functions:

vec4 l_renpy_blur_cubic(float value) {
    vec2 n = vec2(1.0 - value, value);
    vec2 s = n * n;
    vec2 outer = s * n * (1.0 / 6.0);
    vec2 inner = vec2(2.0 / 3.0) - s * (2.0 - n) * 0.5;

    return vec4(outer.x, inner.y, inner.x, outer.y);
}

vec4 l_renpy_blur_sample_bicubic(sampler2D tex, vec2 uv, vec4 size, float lod) {
    vec2 tex_coords = uv * size.xy - 0.5;
    vec2 fraction = fract(tex_coords);
    tex_coords -= fraction;

    vec4 xcubic = l_renpy_blur_cubic(fraction.x);
    vec4 ycubic = l_renpy_blur_cubic(fraction.y);

    vec4 c = tex_coords.xxyy + vec2(-0.5, 1.5).xyxy;
    vec4 s = vec4(xcubic.xz + xcubic.yw, ycubic.xz + ycubic.yw);
    vec4 offset = c + vec4(xcubic.yw, ycubic.yw) / s;

    offset *= size.zzww;

    vec4 sample0 = textureLod(tex, offset.xz, lod);
    vec4 sample1 = textureLod(tex, offset.yz, lod);
    vec4 sample2 = textureLod(tex, offset.xw, lod);
    vec4 sample3 = textureLod(tex, offset.yw, lod);

    return mix(mix(sample3, sample2, s.x), mix(sample1, sample0, s.x), s.z);
}

Fragment shader (priority 200):

fragment_color = l_renpy_blur_sample_bicubic(tex0, v_tex_coord, v_renpy_blur_floor_size, v_renpy_blur_lod_blend.x);

if (v_renpy_blur_lod_blend.y > 0.0) {
    vec4 l_renpy_blur_next_mip = l_renpy_blur_sample_bicubic(tex0, v_tex_coord, v_renpy_blur_ceil_size, v_renpy_blur_lod_blend.x + 1.0);
    fragment_color = mix(fragment_color, l_renpy_blur_next_mip, v_renpy_blur_lod_blend.y);
}

float l_renpy_blur_noise = (fract(52.9829189 * fract(dot(gl_FragCoord.xy, vec2(0.06711056, 0.00583715)))) - 0.5) * (2.0 / 255.0);

fragment_color.rgb = clamp(fragment_color.rgb + vec3(l_renpy_blur_noise * fragment_color.a), vec3(0.0), vec3(fragment_color.a));

textshader.jitter

Variables:

uniform vec2 u_textshader_jitter_jitter;
uniform float u_textshader_jitter_duration;
uniform float u_textshader_jitter_individual;
uniform vec4 u_random;
uniform float u_text_time;
uniform float u_text_to_drawable;
in float a_text_index;

Vertex shader (priority 30):

if (u_textshader_jitter_duration <= 0.0 || u_text_time < u_textshader_jitter_duration) {
    vec2 l_textshader_jitter_random = u_random.xy;

    if (u_textshader_jitter_individual != 0.0) {
        vec2 l_textshader_jitter_seed = vec2(a_text_index, a_text_index + 1.0) + u_random.xy;
        l_textshader_jitter_random = fract(sin(vec2(
            dot(l_textshader_jitter_seed, vec2(12.9898, 78.233)),
            dot(l_textshader_jitter_seed, vec2(39.3468, 11.135))
        )) * 43758.5453);
    }

    vec2 l_textshader_jitter_jitter = u_textshader_jitter_jitter * u_text_to_drawable;
    gl_Position.xy += l_textshader_jitter_jitter * l_textshader_jitter_random - l_textshader_jitter_jitter / 2.0;
}
